import{Types}from"mongoose";\r\nimport{Budget}from"../models/Budget";\r\nimport{Transaction}from"../models/Transaction";\r\nimport{Goal}from"../models/Goal";\r\nimport{Liability}from"../models/Liability";\r\nimport{RecurringTransaction}from"../models/RecurringTransaction";\r\nimport{Notification}from"../models/Notification";\r\n
async function once(userId:Types.ObjectId,type:string,title:string,message:string,actionUrl:string,eventKey:string){await Notification.findOneAndUpdate({userId,type,title,isRead:false},{$setOnInsert:{userId,type,title,message,actionUrl,isRead:false,eventKey}},{upsert:true})}
export async function refreshNotifications(userId:Types.ObjectId){const now=new Date(),from=new Date(now.getFullYear(),now.getMonth(),1),to=new Date(now.getFullYear(),now.getMonth()+1,0,23,59,59),budgets=await Budget.find({userId,month:now.getMonth()+1,year:now.getFullYear()}).populate("categoryId");\r\nconst budgetLabel=(b:any)=>b.scope==="overall"?"Overall spending":(b.categoryId as any).name;\r\nfor(const b of budgets){const match:any={userId,type:"expense",transactionDate:{$gte:from,$lte:to}};\r\nif(b.scope!=="overall")match.categoryId=(b.categoryId as any)._id;\r\nconst r=await Transaction.aggregate([{$match:match},{$group:{_id:null,total:{$sum:"$amount"}}}]),pct=(r[0]?.total||0)/b.limit*100;\r\nif(pct>=100)await once(userId,"budget_exceeded",`${budgetLabel(b)} budget exceeded`,`Spending reached ${pct.toFixed(0)}% of the limit.`,"/budgets",`budget_exceeded:${b._id}`);\r\nelse if(pct>=80)await once(userId,"budget_warning",`${budgetLabel(b)} budget warning`,`Spending reached ${pct.toFixed(0)}% of the limit.`,"/budgets",`budget_warning:${b._id}`)};\r\nconst goals=await Goal.find({userId,status:"active"});\r\nfor(const g of goals){const x=await Transaction.aggregate([{$match:{userId,type:"saving",goalId:g._id}},{$group:{_id:null,total:{$sum:{$cond:[{$eq:["$metadata.activity","withdrawal"]},{$multiply:["$amount",-1]},"$amount"]}}}}]);\r\nif((x[0]?.total||0)>=g.targetAmount)await once(userId,"goal_reached",`${g.name} reached`,`You reached your savings target.`,"/goals",`goal_reached:${g._id}`)};\r\nconst soon=new Date(now.getTime()+7*864e5),bills=await RecurringTransaction.find({userId,isBill:true,isActive:true,nextRunDate:{$gte:now,$lte:soon}});\r\nfor(const b of bills)await once(userId,"bill_due",`${b.title} due soon`,`â‚¹${b.amount.toLocaleString("en-IN")} is due on ${b.nextRunDate.toLocaleDateString("en-IN")}.`,"/recurring",`bill_due:${b._id}:${b.nextRunDate.toISOString()}`);\r\nconst debts=await Liability.find({userId,status:"active",paymentDay:{$exists:true}});\r\nfor(const d of debts){const due=new Date(now.getFullYear(),now.getMonth(),d.paymentDay!);\r\nif(due<now)due.setMonth(due.getMonth()+1);\r\nif(due<=soon)await once(userId,"loan_due",`${d.name} payment due`,`${d.emiAmount?`â‚¹${d.emiAmount.toLocaleString("en-IN")}`:"A payment"} is due on ${due.toLocaleDateString("en-IN")}.`,"/liabilities",`loan_due:${d._id}:${due.toISOString()}`)}}
